-- p1-09 / ISSUES 0013 (23 Nov 2026): sound and music back in the Game Player. The engine's
-- Audio.Init makes its players on the window's card (no Viewport group there) from the game's
-- own Audio folder or the shared kit LIB_SFX/kit (sounds baked by LIB_SFX, chip music).
-- Opened as the Game Player opens a game (GL.Open, no window). Nothing is heard: players counted.
local tP, tS, tC, tN, tHas, tSongs, tTypes, tRoots
global gNoMouse, gV3DHas, bNoAudio
put true into bNoAudio   -- this test is the engine's own way; t80 is LIB_AUDIO
put true into gNoMouse
put "/home/igame3d/projects" into tP
start using stack (tP & "/LIB_DEV/livecode/game_launch.livecodescript")
get GL.Open(tP, "Canyon", false)
if it is not empty then return "FAIL did not open: " & it
put "Play Canyon" into tS
wait 100 milliseconds with messages
dispatch "Player.StopTimer" to stack tS
put the long id of this card of stack tS into tC
put ck("opens with no error", line 1 of the uLastError of stack tS, empty) & return after tLog
put value("Audio.Roots()", tC) into tRoots
put ck("the kit is a root", tRoots contains "LIB_SFX/kit", true) & return after tLog
put ck("players' group on the card", there is a group "AudioPlayers" of tC, true) & return after tLog
put the number of players of group "AudioPlayers" of tC into tN
put ck("players made (47+)", tN >= 47, true) & return after tLog
put value("Audio.Types()", tC) into tTypes
put ck("20 sound types", the number of lines of tTypes, 20) & return after tLog
put ck("Explode is one", "Explode" is among the lines of tTypes, true) & return after tLog
put ck("PlayerLaser has 10 players", there is a player "PlayerLaser10" of tC, true) & return after tLog
put ck("a player has its file", there is a file (the fileName of player "Explode1" of tC), true) & return after tLog
put value("Audio.Songs()", tC) into tSongs
put ck("4 songs (theme kept for the title)", the number of lines of tSongs, 4) & return after tLog
put ck("Engine.Has sound", value("Engine.Has(" & quote & "sound" & quote & ")", tC), true) & return after tLog
put ck("Engine.Has music", value("Engine.Has(" & quote & "music" & quote & ")", tC), true) & return after tLog
put ck("a song is on", value("Audio.CurrentSong()", tC) is among the lines of tSongs, true) & return after tLog
try
   send "Sound.Play" && quote & "Explode" & quote to tC
   send "Sound.Play" && quote & "NoSuchSound" & quote to tC
   put "PASS Sound.Play plays and a missing type is silent" & return after tLog
catch tErr
   put "FAIL Sound.Play threw: " & line 1 of tErr & return after tLog
end try
dispatch "Player.TogglePause" to stack tS
put ck("pause keeps the song", value("Audio.CurrentSong()", tC) is not empty, true) & return after tLog
dispatch "Player.TogglePause" to stack tS
-- a second window: its own players again
close stack tS
get GL.Open(tP, "Hunter", false)
if it is not empty then return tLog & "FAIL Hunter did not open: " & it
put "Play Hunter" into tS
wait 100 milliseconds with messages
dispatch "Player.StopTimer" to stack tS
put the long id of this card of stack tS into tC
put ck("next game: players again", the number of players of group "AudioPlayers" of tC >= 47, true) & return after tLog
close stack tS
put ck("closing stops the music", value("Audio.CurrentSong()"), empty) & return after tLog
put false into bNoAudio
return tLog
